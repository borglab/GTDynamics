cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(contact_initialization_backend LANGUAGES CXX C)
  set(CMAKE_CXX_STANDARD 17)
  include(CMakeFindDependencyMacro)
  find_package(GTSAM REQUIRED)
  enable_testing()
endif()
find_package(Boost 1.75 CONFIG REQUIRED)
get_filename_component(CONTACT_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(gtdynamics_contact_initializer STATIC
  "${CONTACT_REPO_ROOT}/gtdynamics/utils/ContactInitializer.cpp")
target_include_directories(gtdynamics_contact_initializer PUBLIC "${CONTACT_REPO_ROOT}")
target_link_libraries(gtdynamics_contact_initializer PUBLIC gtsam)
target_compile_features(gtdynamics_contact_initializer PUBLIC cxx_std_17)
add_executable(contact_initializer optimizer.cpp)
target_include_directories(contact_initializer PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(contact_initializer PRIVATE gtdynamics_contact_initializer)
target_compile_definitions(contact_initializer PRIVATE CONTACT_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
add_executable(contact_initializer_test test_initializer.cpp)
target_link_libraries(contact_initializer_test PRIVATE gtdynamics_contact_initializer)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(contact_initializer PRIVATE "LINKER:--disable-new-dtags")
  target_link_options(contact_initializer_test PRIVATE "LINKER:--disable-new-dtags")
endif()
add_test(NAME contact_initializer_factors_and_synthetic COMMAND contact_initializer_test)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME contact_initializer_g1_native
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_optimizer.py
            $<TARGET_FILE:contact_initializer>)
endif()
